# ggbot/api/server.py
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.middleware.cors import CORSMiddleware
import asyncio
from typing import Dict
import json

class GGbotAPIServer:
    def __init__(self, runtime: AgentRuntime):
        self.app = FastAPI(title="GGbot API")
        self.runtime = runtime
        self.connections: Dict[str, WebSocket] = {}
        
        # 配置CORS
        self.app.add_middleware(
            CORSMiddleware,
            allow_origins=["*"],
            allow_credentials=True,
            allow_methods=["*"],
            allow_headers=["*"],
        )
        
        # 设置路由
        self.setup_routes()
        
    def setup_routes(self):
        @self.app.websocket("/ws")
        async def websocket_endpoint(websocket: WebSocket):
            await self.handle_websocket(websocket)
            
        @self.app.post("/api/v1/messages")
        async def send_message(message: MessageRequest):
            return await self.handle_message(message)
            
        # ... 其他REST端点
        
    async def handle_websocket(self, websocket: WebSocket):
        """处理WebSocket连接"""
        await websocket.accept()
        connection_id = str(uuid.uuid4())
        self.connections[connection_id] = websocket
        
        try:
            # 订阅事件总线
            def event_handler(event: RuntimeEvent):
                asyncio.create_task(
                    self.send_event(websocket, event)
                )
            
            subscription = subscribe_to_events(event_handler)
            
            # 接收消息循环
            while True:
                data = await websocket.receive_json()
                await self.handle_client_message(data, websocket)
                
        except WebSocketDisconnect:
            pass
        finally:
            # 清理
            get_global_event_bus().unsubscribe(subscription)
            del self.connections[connection_id]
            
    async def send_event(self, websocket: WebSocket, event: RuntimeEvent):
        """发送事件到客户端"""
        message = {
            "type": "event",
            "event_type": event.type,
            "data": event.data,
            "timestamp": time.time()
        }
        await websocket.send_json(message)

// src/services/websocket.ts
class GGbotWebSocketClient {
  private ws: WebSocket | null = null;
  private reconnectAttempts = 0;
  private maxReconnectAttempts = 5;
  private eventHandlers: Map<string, Function[]> = new Map();
  
  async connect(url: string): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(url);
      
      this.ws.onopen = () => {
        this.reconnectAttempts = 0;
        this.emit('connected');
        resolve();
      };
      
      this.ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        this.handleMessage(data);
      };
      
      this.ws.onerror = (error) => {
        this.emit('error', error);
        reject(error);
      };
      
      this.ws.onclose = () => {
        this.emit('disconnected');
        this.reconnect(url);
      };
    });
  }
  
  private handleMessage(data: any): void {
    switch (data.type) {
      case 'event':
        this.emit('event', data);
        // 分发具体事件类型
        this.emit(`event:${data.event_type}`, data.data);
        break;
      case 'response':
        this.emit(`response:${data.id}`, data);
        break;
      case 'error':
        this.emit('error', data);
        break;
    }
  }
  
  sendCommand(type: string, payload: any): Promise<any> {
    return new Promise((resolve, reject) => {
      const id = uuidv4();
      const message = { type: 'command', id, command: type, payload };
      
      // 设置响应超时
      const timeout = setTimeout(() => {
        this.off(`response:${id}`, resolve);
        reject(new Error('Command timeout'));
      }, 30000);
      
      this.once(`response:${id}`, (response) => {
        clearTimeout(timeout);
        resolve(response);
      });
      
      this.ws?.send(JSON.stringify(message));
    });
  }
  
  // 事件发射器/监听器方法
  on(event: string, handler: Function): void { /* ... */ }
  off(event: string, handler: Function): void { /* ... */ }
  emit(event: string, ...args: any[]): void { /* ... */ }
}

6. 关键决策点
6.1 是否复用Claude Code的Ink组件？
建议：部分复用，适当简化
Claude Code的Ink实现非常复杂（25万行代码），我们可以：
1. 复用基础组件：Box、Text、Input等
2. 借鉴架构模式：虚拟列表、状态管理
3. 简化复杂逻辑：去掉不需要的企业级功能
4. 保持兼容性：确保API设计相似
6.2 通信协议选择
建议：WebSocket为主，REST为辅
• WebSocket：实时事件推送（必须）
• REST API：配置管理、会话操作（可选）
• MessagePack：二进制协议优化性能（可选）
6.3 状态同步策略
建议：乐观更新 + 事件溯源
1. UI本地状态：立即响应用户操作
2. 服务器确认：等待后端确认后同步
3. 冲突解决：基于时间戳的最终一致性
4. 离线支持：本地缓存 + 同步队列
7. 风险评估与缓解
7.1 技术风险
• Ink性能问题：终端渲染可能成为瓶颈
◦ 缓解：虚拟列表、增量更新、渲染优化
• WebSocket稳定性：网络波动导致断开
◦ 缓解：自动重连、消息队列、离线模式
• 内存泄漏：长时间运行可能内存增长
◦ 缓解：内存监控、定期清理、消息限制
7.2 兼容性风险
• 终端兼容性：不同终端支持程度不同
◦ 缓解：功能检测、降级方案、广泛测试
• Python版本：依赖特定Python版本
◦ 缓解：版本检查、虚拟环境、容器化
7.3 维护风险
• 代码复杂度：TypeScript + Python双栈
◦ 缓解：清晰接口定义、自动化测试、文档完善
